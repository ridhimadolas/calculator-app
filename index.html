from js import document def append_input(val): display = document.getElementById('display') if display.value == "0" or display.value == "Error": display.value = str(val) else: display.value = display.value + str(val) def clear_display(event=None): display = document.getElementById('display') display.value = "0" def delete_last(event=None): display = document.getElementById('display') if len(display.value) <= 1 or display.value == "Error": display.value = "0" else: display.value = display.value[:-1] def calculate(event=None): display = document.getElementById('display') try: # Standard Python eval calculation result = eval(display.value) display.value = str(result) except Exception: display.value = "Error"
